🔍 السبب الجذري: عند التمرير للأعلى، lastY يتم تحديثه في كل إطار حتى لو كان الفرق صغيراً جداً (1-2 بكسل)، مما يمنع الوصول إلى العتبة المطلوبة (5-10 بكسل) لإظهار الهيدر.
❌ الكود الخاطئ
// مشكلة: lastY يتحدث في كل مرة window.onscroll = function() { var y = window.scrollY; if (y <= 60) { show(); } else if (y > lastY + 10) { hide(); } else if (y < lastY - 5) { show(); } lastY = y; // يتم التحديث دائماً }; // الفرق بين القيم المتتالية يكون 1-2 بكسل فقط // العتبة 5-10 بكسل لا تتحقق أبداً!
✅ الكود الصحيح
// حل: تتبع الذروة (peakY) var peakY = 0; // أعلى نقطة للتمرير للأسفل var lastShownY = 0; // آخر نقطة تم إظهار الهيدر فيها window.addEventListener('scroll', function() { var y = window.scrollY; if (y <= 80) { peakY = 0; lastShownY = 0; show(); } else if (y > peakY) { peakY = y; // تحديث الذروة if (y - lastShownY > 60) hide(); } else { // تمرير لأعلى: قارن بالذروة if (peakY - y > 30) { lastShownY = y; peakY = y; show(); } } });
